	private String translateXmlStringToCsvString(String xmlString) {
		try {
	        DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
	        DocumentBuilder builder = factory.newDocumentBuilder();
	        Document doc = builder.parse(new InputSource(new StringReader(xmlString)));

	        XPath xpath = XPathFactory.newInstance().newXPath();

	        Map<String, List<String>> xpathValues = new LinkedHashMap<>();
	        int maxSize = 0;

	        for (Map.Entry<String, String> entry : consumerCsvMappings.entrySet()) {
	            String header = entry.getKey();
	            String xp = entry.getValue().trim();

	            NodeList nodes = (NodeList) xpath.evaluate(xp, doc, XPathConstants.NODESET);
	            List<String> values = new ArrayList<>();

	            for (int i = 0; i < nodes.getLength(); i++) {
	                values.add(nodes.item(i).getTextContent());
	            }

	            xpathValues.put(header, values);
	            maxSize = Math.max(maxSize, values.size());
	        }

	        StringBuilder csv = new StringBuilder();

	        List<String> headers = new ArrayList<>(consumerCsvMappings.keySet());
	        csv.append(String.join(",", headers)).append("\n");

	        for (int i = 0; i < maxSize; i++) {
	            List<String> row = new ArrayList<>();

	            for (String header : headers) {
	                List<String> values = xpathValues.get(header);
	                row.add(i < values.size() ? String.format("\"%s\"", values.get(i)) : "");
	            }

	            csv.append(String.join(",", row)).append("\n");
	        }
	        return csv.toString();
	        
	    } catch (Exception e) {
	        e.printStackTrace();
	        return "";
	    }

	}

	        XPath xpath = XPathFactory.newInstance().newXPath();

	        for (Map.Entry<String, String> entry : consumerCsvMappings.entrySet()) {
	            String header = entry.getKey();
	            String xp = entry.getValue().trim();

	            List<String> values = new ArrayList<>();

	            for (int i = 0; i < nodes.getLength(); i++) {

	            }

	            xpathValues.put(header, values);
	            maxSize = Math.max(maxSize, values.size());

	        List<Map<String, String>> jsonObjects = new ArrayList<>();

	        for (int i = 0; i < maxSize; i++) {

	        }

	        Gson gson = new GsonBuilder().setPrettyPrinting().create();

	private static String checkDataFormat(String content) {
		try {
			DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
	        DocumentBuilder builder = factory.newDocumentBuilder();
	        builder.parse(new java.io.ByteArrayInputStream(content.getBytes()));
		    return ".xml";
		} catch (Exception e) {}
		try {
		    new JSONObject(new JSONTokener(content));
		    return ".json";
		} catch (Exception e) {}
		try {
			String[] lines = content.split("\\r?\\n");
			if (lines.length < 2) {
	            return null;
	        }
	        for (String line : lines) {
	            if (!line.contains(",")) {
	                return null;
	            }
	        }
	        int columnCount = lines[0].split(",").length;
	        for (String line : lines) {
	            if (line.split(",").length != columnCount) {
	                return null;
	            }
	        }
		    return ".csv";
		} catch (Exception e) {}
		return null;
	}
	
	private Provider readProvider(String path) {
		String extension = "";
		String fileName = Paths.get(path).getFileName().toString();

	    int dotIndex = fileName.lastIndexOf('.');
	    if (dotIndex > 0) {
	        extension = fileName.substring(dotIndex + 1).toLowerCase();
	    }
		
	    switch (extension) {
    	case "xml":	
    		break;
    	case "json":
    		String jsonContent;
			try {
				jsonContent = new String(Files.readAllBytes(Paths.get(path)));
				
	            JSONObject jsonObject = new JSONObject(jsonContent);
	            String xmlString = XML.toString(jsonObject);
	            Path tempFilePath = Files.createTempFile(UUID.randomUUID().toString(), ".xml");
	            Files.write(tempFilePath, xmlString.getBytes());
	            
	            path = tempFilePath.toString();
			} catch (IOException e) {
				e.printStackTrace();
			}

    		break;
    	case "csv":
    		String[] keys, values;
    		try (BufferedReader br = new BufferedReader(new FileReader(path))) {
    			keys = br.readLine().split(",");
    			values = br.readLine().split(",");
    			
    			ObjectMapper mapper = new ObjectMapper();
    		    ObjectNode rootNode = mapper.createObjectNode();
    		     
    		    for (int i = 0; i < keys.length; i++) {
    		    	String[] keyPaths = keys[i].split("/");
					ObjectNode currentNode = rootNode;
					for (int j = 0; j < keyPaths.length - 1; j++) {
					    if (!currentNode.has(keyPaths[j])) {
					        currentNode.putObject(keyPaths[j]);
					    }
					    currentNode = (ObjectNode) currentNode.get(keyPaths[j]);
					}
					currentNode.put(keyPaths[keyPaths.length - 1], values[i]);
    		    }

    		    String jsonOutput = mapper.writerWithDefaultPrettyPrinter().writeValueAsString(rootNode);
    		    
    		    Path tempFilePath = Files.createTempFile(UUID.randomUUID().toString(), ".json");
    		    try (FileWriter fileWriter = new FileWriter(tempFilePath.toString())) {
    	            fileWriter.write(jsonOutput);
    	        }
    		    return readProvider(tempFilePath.toString());
            } catch (IOException e) {
				e.printStackTrace();
			}

    		
    		break;
        default:
            throw new IllegalArgumentException("Invalid File Format <.xml/.json/.csv>: ." + extension);
	    }
	    
	    try {
            File file = new File(path);
            JAXBContext jaxbContext = JAXBContext.newInstance(Provider.class);
            Unmarshaller jaxbUnmarshaller = jaxbContext.createUnmarshaller();
            Provider provider = (Provider) jaxbUnmarshaller.unmarshal(file);
            
            return provider;
        } catch (JAXBException e) {
            e.printStackTrace();
        }
		return null;
	}

	public void exportAsJSON(Object object, String path) {
        Map<String, Object> convertedObject = new HashMap<>();
        for (Field field : object.getClass().getDeclaredFields()) {
            field.setAccessible(true);
            try {
                Object value = field.get(object);
                if (value instanceof XMLGregorianCalendar) {
                    convertedObject.put(field.getName(), ((XMLGregorianCalendar) value).toXMLFormat());
                } else {
                    convertedObject.put(field.getName(), value);
                }
            } catch (IllegalAccessException e) {
                e.printStackTrace();
            }
        }
        
        Gson gson = new GsonBuilder().setPrettyPrinting().create();
        String json = gson.toJson(convertedObject);

        try (FileWriter writer = new FileWriter(path + "/consumer.json")) {
            writer.write(json);
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
	
	private String exportAsXML(Object object, String path) throws JAXBException {
		JAXBContext jaxbContext = JAXBContext.newInstance(object.getClass());
        Marshaller marshaller = jaxbContext.createMarshaller();
        marshaller.setProperty(Marshaller.JAXB_FORMATTED_OUTPUT, true);
        StringWriter sw = new StringWriter();
        marshaller.marshal(object, sw);
        FileWriter writer;
        try {
            writer = new FileWriter(path + "/consumer.xml");
            writer.write(sw.toString());
            writer.close();
        } catch (IOException e) {
            e.printStackTrace();
        }
        return sw.toString();
    }
